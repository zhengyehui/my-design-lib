Can you help me build a REST API endpoint in Python using FastAPI that validates incoming JSON payloads against a Pydantic model and returns structured error responses?

N

Absolutely! Here's a complete FastAPI endpoint with Pydantic validation and structured error handling. I'll include custom exception handlers for a consistent error format:

python fastapi_endpoint.py 
from fastapi import FastAPI, HTTPException, Request
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field, validator
from typing import Optional, List
import uuid

app = FastAPI(title="Neuron API")

# ── Pydantic Models ──
class ItemCreate(BaseModel):
    name: str = Field(..., min_length=1, max_length=128)
    description: Optional[str] = None
    price: float = Field(..., gt=0)
    tags: List[str] = Field(default_factory=list)

    @validator("tags")
    def validate_tags(cls, v):
        if len(v) > 10:
            raise ValueError("Maximum 10 tags allowed")
        return [t.lower().strip() for t in v]

# ── Custom Error Handler ──
@app.exception_handler(422)
async def validation_handler(request, exc):
    errors = []
    for err in exc.detail:
        errors.append({
            "field": " -> ".join(
                str(l) for l in err["loc"]
            ),
            "message": err["msg"],
            "type": err["type"]
        })
    return JSONResponse(
        status_code=422,
        content={"error": "validation_failed",
                 "details": errors}
    )

@app.post("/api/items", status_code=201)
async def create_item(item: ItemCreate):
    return {"id": str(uuid.uuid4()),
            "status": "created", **item.dict()}

The key parts: the ItemCreate model uses Field validators for constraints, and the custom 422 handler reformats Pydantic's default error output into a cleaner structure your frontend can parse easily.

That's great. Now can you add authentication using JWT tokens? I want a /login endpoint that returns an access token and a dependency that protects routes.

N

Here's a JWT authentication layer with a login endpoint and a reusable get_current_user dependency. This uses python-jose for token signing:

python auth.py 
from datetime import datetime, timedelta
from jose import JWTError, jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer

SECRET_KEY = "your-secret-key-here"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 30

oauth2_scheme = OAuth2PasswordBearer(
    tokenUrl="/login"
)

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(
        minutes=ACCESS_TOKEN_EXPIRE_MINUTES
    )
    to_encode.update({"exp": expire})
    return jwt.encode(
        to_encode, SECRET_KEY, algorithm=ALGORITHM
    )

async def get_current_user(
    token: str = Depends(oauth2_scheme)
):
    credentials_exc = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Invalid or expired token",
        headers={"WWW-Authenticate": "Bearer"}
    )
    try:
        payload = jwt.decode(
            token, SECRET_KEY, algorithms=[ALGORITHM]
        )
        user_id: str = payload.get("sub")
        if user_id is None:
            raise credentials_exc
        return {"user_id": user_id}
    except JWTError:
        raise credentials_exc

To protect a route, just add user = Depends(get_current_user) to your endpoint signature. The dependency will automatically return 401 if the token is missing, malformed, or expired.
