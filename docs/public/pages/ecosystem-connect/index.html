  • ⚡ RESTful API with GraphQL support

terminal — nexus-sdk
// Install the NexusConnect SDK
$ npm install @nexusconnect/sdk
 
// Initialize the client
import { NexusClient } from '@nexusconnect/sdk'
 
const nexus = new NexusClient({
  apiKey: 'nx_live_...',
  workspace: 'my-app'
})
 
// Create an integration
const conn = await nexus.integrations.connect({
  provider: 'salesforce',
  scopes: ['read', 'write']
})
 
console.log(conn.status) // "connected"
